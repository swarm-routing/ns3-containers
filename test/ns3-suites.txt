# Modules whose QUICK unit suites test/unit.sh builds and runs (ns3-containers#40).
#
#   <group> <module> ...
#
# `full` runs on the debug and sanitized images (what CI consumes; asserts on),
# `light` on default and optimized. UNIT_SET=full|light overrides the profile:
# on pull requests the Debian, slim and Alpine images run `light` (#87). The MANET stack the org's protocols use is
# in `full`: core, network, internet, mobility, wifi, propagation, the routing
# models (aodv, olsr, dsdv) and mesh (in NS3_MODULES).
full  core network internet mobility wifi propagation aodv olsr dsdv mesh
light core network internet

# Suites deferred to main (ns3-containers#86, #87; maintainer decision on #86,
# 2026-11-10). `main-only <suite>...`: on a pull request (UNIT_PR=1) the unit
# stage leaves these suites out, whatever the group, and says so in the job
# log and the step summary. Pushes to main, schedule and dispatch runs, and
# any run without UNIT_PR run them. Nothing is tagged on a pull request and
# promote runs only on main, so every published image is still tested with
# them (golden rule 17).
#
# ns3-tcp-cubic (internet, so in both groups) is the one suite that sets the
# stage's wall time: 1343-1936 s under ASan+UBSan, 470 s on debug and 348 s on
# the light group's Debian images, where it is 348 of the 350 s of test.py
# (#89's and #92's runs).
main-only ns3-tcp-cubic

# Split test jobs (ns3-containers#86). `split <profile> <N>`: publish-images.yml
# tests that profile's image in parallel jobs on the same digest: one job with
# every stage but unit, and the unit stage as N jobs. N = 1 runs the whole
# group in one job; N >= 2 runs `shard <group> i/N <module>...` lines, which
# must list every module of the group exactly once (tools/ns3_matrix.py
# --check-suites, and test/unit.sh at run time).
#
# Sanitized, ns-3.49, measured in #89's run: the stages but unit take 712 s,
# the unit stage 2475 s (test libraries 509 s, test.py 1963 s). test.py's wall
# time is one suite: ns3-tcp-cubic (internet) runs 1936 s alone under
# ASan+UBSan, while the other 212 suites (5022 s in all) share the other three
# cores beside it. No unit shard can finish below that suite: an internet
# shard would still take about 36 min, for about 23 more runner-minutes. So
# the unit stage stays one job and only the other stages move out. With
# ns3-tcp-cubic deferred to main on pull requests (#97's run 42914844852) the
# unit job still took 2870 s (test libraries 657 s, test.py 2181 s, whose
# floor is now wifi-mac-ofdma at 1442 s) and the other stages 763 s, on a
# runner about 1.7x slower than #89's: in one job the two would far exceed
# 25 min on any runner, so the split stays.
split sanitized 1
